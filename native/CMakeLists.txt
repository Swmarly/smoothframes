cmake_minimum_required(VERSION 3.24)
project(SmoothFramesNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
include(FetchContent)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(minhook
  GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
  GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
FetchContent_MakeAvailable(minhook)
add_library(SmoothFramesHook SHARED Hook.cpp)
target_link_libraries(SmoothFramesHook PRIVATE minhook d3d11 d3d9 dxgi opengl32)
add_executable(SmoothFramesInjector Injector.cpp)
add_executable(SmoothFramesSmoke Smoke.cpp)
target_link_libraries(SmoothFramesSmoke PRIVATE d3d11 dxgi)
foreach(target SmoothFramesHook SmoothFramesInjector SmoothFramesSmoke)
  target_compile_options(${target} PRIVATE /W4 /permissive-)
endforeach()
enable_testing()
add_test(NAME native-integration COMMAND SmoothFramesSmoke)
set_tests_properties(native-integration PROPERTIES TIMEOUT 40)
install(TARGETS SmoothFramesHook SmoothFramesInjector RUNTIME DESTINATION .)
install(FILES "${minhook_SOURCE_DIR}/LICENSE.txt" DESTINATION . RENAME MinHook-LICENSE.txt)
